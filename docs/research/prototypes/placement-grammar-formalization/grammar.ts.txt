import { parse } from '../../../../src/parse';
import type { OutlineNode } from '../../../../src/model';

export type Label = 'h1' | 'h2' | 'h3' | 'h4' | 'P' | 'I' | 'I#' | 'C' | 'Q';
export interface T {
  l: Label;
  c: T[];
}
export const isH = (l: Label) => l[0] === 'h';
export const lvl = (l: Label) => Number(l.slice(1));
const HEADINGS: Label[] = ['h1', 'h2', 'h3'];

// ---------------------------------------------------------------- generator
// Each nonterminal of the grammar is a function from a size to the forests of
// exactly that size it derives. `col` carries the opening-margin state.

/** Section(n): Content* Heading*, no (P, I) pair, heading levels > n and
 * non-increasing. `prev` and `maxH` are the sibling-sequence automaton's
 * state. */
export function* section(k: number, n: number, col: number, prev: Label | undefined, maxH: number, seenH: boolean): Generator<T[]> {
  if (k === 0) {
    yield [];
    return;
  }
  for (let s = 1; s <= k; s++) {
    // content child
    if (!seenH) {
      for (const t of sectionContent(s, col, prev)) {
        for (const rest of section(k - s, n, col, t.l, maxH, false)) yield [t, ...rest];
      }
    }
    // heading child
    for (const h of HEADINGS) {
      const L = lvl(h);
      if (L <= n || L > maxH) continue;
      for (const kids of section(s - 1, L, col, undefined, Infinity, false)) {
        const t: T = { l: h, c: kids };
        for (const rest of section(k - s, n, col, h, L, true)) yield [t, ...rest];
      }
    }
  }
}
export function* sectionContent(s: number, col: number, prev: Label | undefined): Generator<T> {
  if (s === 1) {
    yield { l: 'C', c: [] };
    if (col <= 3) yield { l: 'Q', c: [] };
  }
  for (const kids of items(s - 1, col)) yield { l: 'P', c: kids };
  if (prev !== 'P') for (const t of item(s, col)) yield t;
}
/** I*: a paragraph's children. */
function* items(k: number, col: number): Generator<T[]> {
  if (k === 0) {
    yield [];
    return;
  }
  for (let s = 1; s <= k; s++)
    for (const t of item(s, col)) for (const rest of items(k - s, col)) yield [t, ...rest];
}
/** Item: item[(Item | Leaf | Atom)*] */
function* item(s: number, col: number): Generator<T> {
  for (const kids of itemChildren(s - 1, col + 2)) yield { l: 'I', c: kids };
}
function* itemChildren(k: number, col: number): Generator<T[]> {
  if (k === 0) {
    yield [];
    return;
  }
  for (let s = 1; s <= k; s++) {
    const heads: T[] = [];
    if (s === 1) {
      heads.push({ l: 'P', c: [] }, { l: 'C', c: [] });
      if (col <= 3) heads.push({ l: 'Q', c: [] });
    }
    const all: Iterable<T> = s === 1 ? [...heads, ...item(1, col)] : item(s, col);
    for (const t of all) for (const rest of itemChildren(k - s, col)) yield [t, ...rest];
  }
}

// ---------------------------------------------------------------- writer / reader
export function write(f: T[]): string {
  const out: string[] = [];
  let k = 0;
  const w = (t: T, col: number): void => {
    const pad = ' '.repeat(col);
    const id = `x${k++}`;
    if (t.l === 'P') out.push(pad + id);
    else if (t.l === 'I') out.push(`${pad}- ${id}`);
    else if (t.l === 'I#') out.push(`${pad}- ## ${id}`);
    else if (t.l === 'C') out.push(pad + '```', pad + id, pad + '```');
    else if (t.l === 'Q') out.push(`${pad}> ${id}`);
    else out.push(`${pad}${'#'.repeat(lvl(t.l))} ${id}`);
    out.push('');
    for (const c of t.c) w(c, t.l === 'I' || t.l === 'I#' ? col + 2 : col);
  };
  for (const t of f) w(t, 0);
  return out.join('\n');
}
export function label(n: OutlineNode): string {
  if (n.kind === 'list-item' && /^\s*[-*+] #/.test(n.lines[0] ?? '')) return 'I#';
  if (n.kind === 'heading') return `h${n.level}`;
  return { paragraph: 'P', 'list-item': 'I', code: 'C', quote: 'Q' }[n.kind as string] ?? `?${n.kind}`;
}
export function sameShape(f: T[], ns: readonly OutlineNode[]): boolean {
  return f.length === ns.length && f.every((t, i) => t.l === label(ns[i]!) && sameShape(t.c, ns[i]!.children));
}
