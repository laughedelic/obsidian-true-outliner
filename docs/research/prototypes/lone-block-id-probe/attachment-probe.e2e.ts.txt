/**
 * The metadata probe behind docs/research/lone-block-id: which block a lone
 * `^id` line names, for the shapes `zoom-scoped-backlinks` did not probe.
 *
 * To re-run it: copy to `e2e/specs/99-probe-lone-block-id.e2e.ts` and
 *   xvfb-run -a npm run test:e2e:narrow -- 99-probe-lone-block-id
 * It asserts nothing — it prints one PROBE_RESULT line per shape and is kept
 * out of the suite for that reason.
 *
 * Each shape is a note of its own, so no shape's id can name another's block.
 * For each id it reports the lines `CachedMetadata.blocks` gives it, the
 * sections the note splits into, and the text of the named range's first and
 * last lines.
 */

describe('probe: lone block ids', function () {
  before(async function () {
    await obsidianPage.resetVault();
    await h.resetPluginState();
  });

  it('measures what each lone id names', async function () {
    const paths: Record<string, string> = {};
    let n = 0;
    for (const [name, md] of Object.entries(SHAPES)) {
      const path = `Scratch/Lone id ${String(++n).padStart(2, '0')}.md`;
      paths[name] = path;
      await h.createNote(path, md);
    }
    await h.waitForMetadataCache();

    for (const [name, path] of Object.entries(paths)) {
      const result = await browser.executeObsidian(
        ({ app }, p, md) => {
          const file = app.vault.getAbstractFileByPath(p) as never;
          const cache = app.metadataCache.getFileCache(file)!;
          const lines = (md as string).split('\n');
          const blocks = Object.values(cache.blocks ?? {}).map((b) => {
            const block = b as { id: string; position: { start: { line: number }; end: { line: number } } };
            const { start, end } = block.position;
            return {
              id: block.id,
              lines: `${start.line}-${end.line}`,
              first: lines[start.line],
              last: lines[end.line],
            };
          });
          const sections = (cache.sections ?? []).map(
            (s) => `${s.type}@${s.position.start.line}-${s.position.end.line}`,
          );
          return { blocks, sections };
        },
        path,
        SHAPES[name]!,
      );
      console.log(`PROBE_RESULT ${JSON.stringify({ shape: name, md: SHAPES[name], ...result })}`);
    }
  });
});
