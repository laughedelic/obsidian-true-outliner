// Flush seams in the corpus whose upper block ends in a quote, a callout or a
// list item and whose lower block is not a heading, a fence or a list item.
// Copy to `tests/corpus-seams.test.ts`; writes to CORPUS_OUT.
import { it } from 'vitest';
import { writeFileSync, readFileSync } from 'node:fs';
import { execSync } from 'node:child_process';
import { parse } from '../src/parse';
import type { OutlineNode } from '../src/model';

const final = (n: OutlineNode): OutlineNode => (n.children.length ? final(n.children[n.children.length - 1]!) : n);

it('flush lazy-claimable seams the corpus holds', () => {
  const files = execSync("git -c core.quotepath=off ls-files 'tests/corpus/*.md' 'test-vault/*.md' 'test-vault/**/*.md'")
    .toString().trim().split('\n');
  let seams = 0;
  const rows: string[] = [];
  for (const f of files) {
    const walk = (nodes: readonly OutlineNode[]): void => {
      for (let i = 0; i < nodes.length; i++) {
        walk(nodes[i]!.children);
        if (i === nodes.length - 1) continue;
        seams++;
        const leaf = final(nodes[i]!);
        const next = nodes[i + 1]!;
        if (leaf.trailingGap.length > 0) continue;
        if (!['quote', 'callout', 'list-item'].includes(leaf.kind)) continue;
        if ((next.kind === 'heading' && next.setext !== true) || next.kind === 'code' || next.kind === 'list-item') continue;
        rows.push(`${f}: ${JSON.stringify(leaf.lines.at(-1))} >> ${JSON.stringify(next.lines[0])}`);
      }
    };
    walk(parse(readFileSync(f, 'utf8')).children);
  }
  writeFileSync(process.env.CORPUS_OUT ?? '/tmp/corpus-seams.txt', `files ${files.length}, sibling seams ${seams}, flagged ${rows.length}\n${rows.join('\n')}\n`);
});
