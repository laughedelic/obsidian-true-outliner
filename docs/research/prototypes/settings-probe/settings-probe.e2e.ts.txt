/**
 * The probe behind docs/research/settings-tab-structure.md: measures and
 * screenshots the current settings tab, then every candidate layout under
 * `layouts/` (see its README for the format), installed at runtime through the
 * tab's own `getSettingDefinitions()` and `update()`: the top level, every page,
 * and what the settings search index makes of it.
 *
 * To run it: copy to `e2e/specs/99-settings-probe.e2e.ts`, then
 *   PROBE_OUT=/tmp/settings-probe PROBE_PLATFORM=desktop OBSIDIAN_VERSION=1.13.7 \
 *     sh e2e/docker/start-xvfb-and-run.sh npm run test:e2e:narrow -- 99-settings-probe
 * and again with `PROBE_PLATFORM=mobile` and `--mobile`; then
 *   node docs/research/prototypes/settings-probe/gallery.mjs /tmp/settings-probe
 * folds both runs into one `gallery.html`. It asserts nothing and is kept out of
 * the suite for that reason.
 */

import { browser } from '@wdio/globals';
import * as fs from 'node:fs';
import * as path from 'node:path';

const OUT = process.env.PROBE_OUT ?? '/tmp/settings-probe';
const PLATFORM = process.env.PROBE_PLATFORM ?? 'desktop';
const LAYOUTS = process.env.PROBE_LAYOUTS ?? 'docs/research/prototypes/settings-probe/layouts';
const DIR = path.join(OUT, PLATFORM);
fs.mkdirSync(DIR, { recursive: true });

interface Layout {
  title: string;
  keepDesc?: boolean;
  items: unknown[];
}

const layouts: { id: string; layout: Layout }[] = fs
  .readdirSync(LAYOUTS)
  .filter((f) => f.endsWith('.json'))
  .sort()
  .map((f) => ({ id: f.replace(/\.json$/, ''), layout: JSON.parse(fs.readFileSync(path.join(LAYOUTS, f), 'utf8')) }));

async function shot(name: string): Promise<void> {
  await browser.pause(400);
  await browser.saveScreenshot(path.join(DIR, `${name}.png`));
}

function record(name: string, data: unknown): void {
  fs.writeFileSync(path.join(DIR, `${name}.json`), JSON.stringify(data, null, 2));
}

async function openTab(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const a = app as any;
    // The 1.13 desktop default opens settings in a popout window, which a
    // screenshot of the main window never sees.
    a.vault.setConfig('settingsPopoutWindow', false);
    a.setting.open();
    a.setting.openTabById('true-outliner');
  });
  await browser.pause(500);
}

async function closeSettings(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    (app as any).setting.close();
  });
  await browser.pause(300);
}

async function scrollToBottom(): Promise<boolean> {
  return browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const page = setting.pageStack?.[setting.pageStack.length - 1]?.page;
    const tab = setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    const content = page
      ? (page.rootEl.querySelector('.setting-page-content') ?? page.rootEl)
      : (tab.containerEl.closest('.vertical-tab-content') ?? tab.containerEl);
    const scrolls = content.scrollHeight > content.clientHeight + 4;
    content.scrollTop = content.scrollHeight;
    return scrolls;
  });
}

/** Geometry of what is open: the tab, or the page on top of the stack. */
async function measure(): Promise<unknown> {
  return browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const page = setting.pageStack?.[setting.pageStack.length - 1]?.page;
    const tab = setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    const root: HTMLElement = page ? page.rootEl : tab.containerEl;
    const content = page
      ? (root.querySelector('.setting-page-content') ?? root)
      : (root.closest('.vertical-tab-content') ?? root);
    const rows = Array.from(root.querySelectorAll('.setting-item')) as HTMLElement[];
    return {
      title: page ? page.title : null,
      viewport: { w: window.innerWidth, h: window.innerHeight },
      clientHeight: content.clientHeight,
      scrollHeight: content.scrollHeight,
      viewports: +(content.scrollHeight / content.clientHeight).toFixed(2),
      rows: rows.map((r) => {
        const desc = r.querySelector('.setting-item-description') as HTMLElement | null;
        const descH = desc ? desc.getBoundingClientRect().height : 0;
        const lh = desc ? parseFloat(getComputedStyle(desc).lineHeight) : 0;
        return {
          name: (r.querySelector('.setting-item-name') as HTMLElement | null)?.textContent ?? '',
          kind: r.classList.contains('setting-item-heading')
            ? 'heading'
            : r.classList.contains('mod-navigable')
              ? 'page'
              : 'row',
          h: +r.getBoundingClientRect().height.toFixed(1),
          descLines: lh ? Math.round(descH / lh) : 0,
        };
      }),
    };
  });
}

/** Search results for a query, as the index resolves them for our tab. */
async function search(query: string): Promise<unknown> {
  return browser.executeObsidian(({ app }, q: string) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const groups = (app as any).setting.searchIndex.search(q) as any[];
    return groups
      .filter((g) => g.tab?.id === 'true-outliner')
      .map((g) => ({ page: g.page?.name ?? null, results: g.results.map((r: any) => r.entry.definition.name) }));
  }, query);
}

/** Installs a layout: the JSON is turned into definitions inside Obsidian,
 * against the tab's own current rows so every control keeps its binding. */
async function install(layout: Layout): Promise<void> {
  await browser.executeObsidian(
    ({ app }, spec: Layout) => {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
      if (!tab.__original) tab.__original = tab.getSettingDefinitions.bind(tab);
      const rows: any[] = tab.__original();
      const byKey: Record<string, any> = {};
      for (const r of rows) if (r.control) byKey[r.control.key] = r;
      const preview = rows.find((r) => r.render);
      const firstSentence = (s: string): string => {
        const m = /^(.*?[.!?])(\s|$)/.exec(s);
        return m ? m[1] : s;
      };
      const data = () => tab.plugin.data;
      const valueOf = (key: string) => (): string => {
        const r = byKey[key];
        const v = data()[key];
        if (!r) return '';
        if (r.control.type === 'toggle') return v ? 'On' : 'Off';
        return r.control.options?.[v] ?? String(v);
      };
      const build = (item: any): any => {
        if (item === 'preview') return { ...preview };
        if (typeof item === 'string') item = { key: item };
        if (item.group) return { type: 'group', heading: item.group, items: item.items.map(build) };
        if (item.page) {
          return {
            type: 'page',
            name: item.page,
            desc: item.desc ?? '',
            ...(item.value ? { displayValue: valueOf(item.value) } : {}),
            items: item.items.map(build),
          };
        }
        const r = byKey[item.key];
        if (!r) throw new Error(`no setting ${item.key}`);
        const desc = typeof r.desc === 'string' ? r.desc : (r.desc?.textContent ?? '');
        return {
          ...r,
          name: item.name ?? r.name,
          desc: item.desc ?? (spec.keepDesc ? desc : firstSentence(desc)),
          ...(item.aliases ? { aliases: item.aliases } : {}),
        };
      };
      const defs = spec.items.map(build);
      tab.getSettingDefinitions = () => defs;
      tab.update();
    },
    layout,
  );
}

async function restore(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    if (tab.__original) {
      tab.getSettingDefinitions = tab.__original;
      delete tab.__original;
      tab.update();
    }
  });
}

/** The page entries visible at the current level, by name. */
async function pageNames(): Promise<string[]> {
  return browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const page = setting.pageStack?.[setting.pageStack.length - 1]?.page;
    const tab = setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    const root: HTMLElement = page ? page.rootEl : tab.containerEl;
    return (Array.from(root.querySelectorAll('.setting-item.mod-navigable')) as HTMLElement[]).map(
      (r) => r.querySelector('.setting-item-name')?.textContent ?? '',
    );
  });
}

async function openPage(name: string): Promise<void> {
  await browser.executeObsidian(({ app }, n: string) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const page = setting.pageStack?.[setting.pageStack.length - 1]?.page;
    const tab = setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    const root: HTMLElement = page ? page.rootEl : tab.containerEl;
    const rows = Array.from(root.querySelectorAll('.setting-item.mod-navigable')) as HTMLElement[];
    rows.find((r) => r.querySelector('.setting-item-name')?.textContent === n)?.click();
  }, name);
  await browser.pause(600);
}

async function closePage(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    (app as any).setting.closePage?.();
  });
  await browser.pause(500);
}

const slug = (s: string): string => s.toLowerCase().replace(/[^a-z0-9]+/g, '-');

/** Screenshots and measures the current level, then every page under it. */
async function capture(prefix: string, summary: Record<string, unknown>): Promise<void> {
  await shot(`${prefix}-top`);
  const m = (await measure()) as { viewports: number; rows: unknown[] };
  summary[prefix] = m;
  if (await scrollToBottom()) await shot(`${prefix}-bottom`);
  for (const name of await pageNames()) {
    await openPage(name);
    await capture(`${prefix}-${slug(name)}`, summary);
    await closePage();
  }
}

describe('settings probe', function () {
  this.timeout(300000);

  it('measures the current tab', async function () {
    await openTab();
    const summary: Record<string, unknown> = {};
    await capture('current', summary);
    record('current', { title: 'Current', platform: PLATFORM, pages: summary, search: { guide: await search('guide'), backlinks: await search('backlinks') } });
    await closeSettings();
  });

  for (const { id, layout } of layouts) {
    it(`measures layout ${id}`, async function () {
      await openTab();
      await install(layout);
      await closeSettings();
      await openTab();
      const summary: Record<string, unknown> = {};
      await capture(id, summary);
      record(id, { title: layout.title, platform: PLATFORM, pages: summary, search: { guide: await search('guide'), backlinks: await search('backlinks') } });
      await restore();
      await closeSettings();
    });
  }
});
