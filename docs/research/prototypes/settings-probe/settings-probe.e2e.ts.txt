/**
 * The probe behind docs/research/settings-tab-structure.md: screenshots and
 * measures the current settings tab, then injects two prototype layouts through
 * the tab's own `getSettingDefinitions()` and `update()` and measures those,
 * plus what the settings search index makes of each.
 *
 * To re-run it: copy to `e2e/specs/99-settings-probe.e2e.ts`, then
 *   PROBE_OUT=/tmp/settings-probe PROBE_PLATFORM=desktop \
 *     sh e2e/docker/start-xvfb-and-run.sh npm run test:e2e:narrow -- 99-settings-probe
 * and again with `PROBE_PLATFORM=mobile` and `--mobile`. It asserts nothing and
 * writes PNGs and JSON under PROBE_OUT, which is why it is kept out of the suite.
 */

import { browser } from '@wdio/globals';
import * as fs from 'node:fs';
import * as path from 'node:path';

const OUT = process.env.PROBE_OUT ?? '/tmp/settings-probe';
const PLATFORM = process.env.PROBE_PLATFORM ?? 'desktop';
const DIR = path.join(OUT, PLATFORM);
fs.mkdirSync(DIR, { recursive: true });

async function shot(name: string): Promise<void> {
  await browser.pause(400);
  await browser.saveScreenshot(path.join(DIR, `${name}.png`));
}

function record(name: string, data: unknown): void {
  fs.writeFileSync(path.join(DIR, `${name}.json`), JSON.stringify(data, null, 2));
}

async function openTab(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    // The 1.13 desktop default opens settings in a popout window, which a
    // screenshot of the main window never sees.
    (app as any).vault.setConfig('settingsPopoutWindow', false);
    setting.open();
    setting.openTabById('true-outliner');
  });
  await browser.pause(500);
}

async function closeSettings(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    (app as any).setting.close();
  });
  await browser.pause(300);
}

/** Geometry of the open tab: how many viewports it scrolls, and every row. */
async function measureTab(): Promise<unknown> {
  return browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const tab = setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    const content = tab.containerEl.closest('.vertical-tab-content') ?? tab.containerEl;
    const rows = Array.from(tab.containerEl.querySelectorAll('.setting-item')) as HTMLElement[];
    const lh = parseFloat(getComputedStyle(document.body).lineHeight) || 0;
    return {
      viewport: { w: window.innerWidth, h: window.innerHeight },
      content: { clientHeight: content.clientHeight, scrollHeight: content.scrollHeight },
      viewports: +(content.scrollHeight / content.clientHeight).toFixed(2),
      rowCount: rows.length,
      rows: rows.map((r) => {
        const desc = r.querySelector('.setting-item-description') as HTMLElement | null;
        const descH = desc ? desc.getBoundingClientRect().height : 0;
        const descLh = desc ? parseFloat(getComputedStyle(desc).lineHeight) : lh;
        return {
          name: (r.querySelector('.setting-item-name') as HTMLElement | null)?.textContent ?? '',
          cls: r.className,
          h: +r.getBoundingClientRect().height.toFixed(1),
          descH: +descH.toFixed(1),
          descLines: descLh ? Math.round(descH / descLh) : 0,
        };
      }),
    };
  });
}

/** Search results for a query, as the index resolves them: tab, page, item names. */
async function search(query: string): Promise<unknown> {
  return browser.executeObsidian(({ app }, q: string) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const groups = setting.searchIndex.search(q) as any[];
    return groups
      .filter((g) => g.tab?.id === 'true-outliner')
      .map((g) => ({
        tab: g.tab.name,
        page: g.page?.name ?? null,
        results: g.results.map((r: any) => r.entry.definition.name),
      }));
  }, query);
}

/** The first sentence of a description, for the prototypes. */
function firstSentence(s: string): string {
  const m = /^(.*?[.!?])(\s|$)/.exec(s);
  return m ? m[1]! : s;
}

type Def = Record<string, unknown>;

/** Builds prototype definitions from the tab's own current rows, keyed. */
async function installPrototype(variant: 'A' | 'B'): Promise<void> {
  await browser.executeObsidian(
    ({ app }, v: string, first: string) => {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      const setting = (app as any).setting;
      const tab = setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
      // eslint-disable-next-line no-new-func
      const firstSentence = new Function('s', first) as (s: string) => string;
      if (!tab.__original) tab.__original = tab.getSettingDefinitions.bind(tab);
      const rows: any[] = tab.__original();
      const byKey: Record<string, any> = {};
      for (const r of rows) if (r.control) byKey[r.control.key] = r;
      const row = (key: string, name?: string): any => {
        const r = byKey[key];
        if (!r) throw new Error(`no row ${key}`);
        const desc = typeof r.desc === 'string' ? r.desc : r.desc?.textContent ?? '';
        return { ...r, name: name ?? r.name, desc: firstSentence(desc) };
      };
      const preview = rows.find((r) => r.render);
      const data = () => tab.plugin.data;
      const onOff = (k: string) => () => (data()[k] ? 'On' : 'Off');

      const guides = [
        row('guideVisibility', 'Which guides to draw'),
        row('guideIntensity', 'Strength'),
        row('guideHighlight', 'Highlight at the cursor'),
        row('guideHideSingleRoot', 'Hide the outermost guide under a single root'),
      ];
      const markers = [
        row('markerVisibility', 'Which nodes get a marker'),
        row('headingMarkerGlyph', 'Heading glyph'),
        row('headingMarkerLevel', 'Heading level'),
        { ...preview },
        row('markerHighlight', 'Highlight at the cursor'),
      ];
      const backlinks = [
        row('backlinksFooter', 'Show backlinks below notes'),
        row('backlinksSuppressCore', 'Hide Obsidian’s own backlinks section'),
        {
          type: 'group',
          heading: 'Amount',
          items: [row('backlinksOverallCap', 'References to show'), row('backlinksGroupHeight', 'Height of one note’s references')],
        },
        {
          type: 'group',
          heading: 'Lineage rows',
          items: [row('backlinksSegmentIcons', 'Markers'), row('backlinksSeparator', 'Separator'), row('backlinksGuides', 'Guide lines')],
        },
      ];
      const top = [row('outlineByDefault'), row('statusBarMode'), row('rememberFolds')];
      const layout = [row('outlineUnit'), row('hideGapLines')];
      const advanced = [row('debugCrossCheck', 'Cross-check the parser against the metadata cache')];

      const defsA = [
        ...top,
        { type: 'group', heading: 'Layout', items: layout },
        { type: 'group', heading: 'Guides', items: guides },
        { type: 'group', heading: 'Markers', items: markers },
        {
          type: 'page',
          name: 'Backlinks',
          desc: 'The structured backlinks footer below a note.',
          displayValue: onOff('backlinksFooter'),
          items: backlinks,
        },
        { type: 'group', heading: 'Advanced', items: advanced },
      ];
      const defsB = [
        ...top,
        {
          type: 'group',
          heading: 'Appearance',
          items: [
            ...layout,
            { type: 'page', name: 'Guides', desc: 'The vertical lines that connect a node to its ancestors.', displayValue: () => ({ all: 'Every level', ancestors: 'Ancestors', subtree: 'Subtree', off: 'Off' })[data().guideVisibility as string], items: guides },
            { type: 'page', name: 'Markers', desc: 'The icon at the start of a node.', displayValue: () => ({ all: 'All', 'with-children': 'Branches', 'headings-and-paragraphs': 'Headings and paragraphs' })[data().markerVisibility as string], items: markers },
          ],
        },
        {
          type: 'group',
          heading: 'Features',
          items: [
            { type: 'page', name: 'Backlinks', desc: 'The structured backlinks footer below a note.', displayValue: onOff('backlinksFooter'), items: backlinks },
          ],
        },
        { type: 'group', heading: 'Advanced', items: advanced },
      ];
      tab.getSettingDefinitions = () => (v === 'A' ? defsA : defsB);
      tab.update();
    },
    variant,
    firstSentence.toString().replace(/^[^{]*\{/, '').replace(/\}$/, ''),
  );
}

async function restoreTab(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    if (tab.__original) {
      tab.getSettingDefinitions = tab.__original;
      delete tab.__original;
      tab.update();
    }
  });
}

async function openPage(name: string): Promise<boolean> {
  const ok = await browser.executeObsidian(({ app }, n: string) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
    const rows = Array.from(tab.containerEl.querySelectorAll('.setting-item.mod-navigable')) as HTMLElement[];
    const row = rows.find((r) => r.querySelector('.setting-item-name')?.textContent === n);
    if (!row) return false;
    row.click();
    return true;
  }, name);
  await browser.pause(600);
  return ok;
}

async function measurePage(): Promise<unknown> {
  return browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const page = setting.pageStack?.[setting.pageStack.length - 1]?.page;
    if (!page) return null;
    const el = page.rootEl as HTMLElement;
    const content = el.querySelector('.setting-page-content') ?? el;
    return {
      title: page.title,
      clientHeight: content.clientHeight,
      scrollHeight: content.scrollHeight,
      rows: (Array.from(el.querySelectorAll('.setting-item')) as HTMLElement[]).map((r) => ({
        name: r.querySelector('.setting-item-name')?.textContent ?? '',
        h: +r.getBoundingClientRect().height.toFixed(1),
        cls: r.className,
      })),
    };
  });
}

async function closePage(): Promise<void> {
  await browser.executeObsidian(({ app }) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    (app as any).setting.closePage?.();
  });
  await browser.pause(500);
}

async function typeSearch(q: string): Promise<void> {
  await browser.executeObsidian(({ app }, query: string) => {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const setting = (app as any).setting;
    const input = setting.searchComponent?.inputEl as HTMLInputElement | undefined;
    if (!input) throw new Error('no search input');
    input.value = query;
    input.dispatchEvent(new Event('input', { bubbles: true }));
  }, q);
  await browser.pause(600);
}

describe('settings probe', function () {
  this.timeout(120000);

  it('measures the current tab', async function () {
    await openTab();
    await shot('current-top');
    record('current', await measureTab());
    await browser.executeObsidian(({ app }) => {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
      const content = tab.containerEl.closest('.vertical-tab-content') ?? tab.containerEl;
      content.scrollTop = content.scrollHeight;
    });
    await shot('current-bottom');
    record('search-current-guide', await search('guide'));
    record('search-current-backlinks', await search('backlinks'));
    await typeSearch('guide');
    await shot('current-search-guide');
    await closeSettings();
  });

  for (const variant of ['A', 'B'] as const) {
    it(`measures prototype ${variant}`, async function () {
      await openTab();
      await installPrototype(variant);
      await closeSettings();
      await openTab();
      await shot(`proto${variant}-top`);
      record(`proto${variant}`, await measureTab());
      record(`search-proto${variant}-guide`, await search('guide'));
      record(`search-proto${variant}-backlinks`, await search('backlinks'));
      await typeSearch('backlinks');
      await shot(`proto${variant}-search-backlinks`);
      await typeSearch('');
      await openTab();
      if (await openPage('Backlinks')) {
        await shot(`proto${variant}-page-backlinks`);
        record(`proto${variant}-page-backlinks`, await measurePage());
        await closePage();
      }
      if (variant === 'B' && (await openPage('Guides'))) {
        await shot('protoB-page-guides');
        record('protoB-page-guides', await measurePage());
        await closePage();
      }
      await restoreTab();
      await closeSettings();
    });
  }

  it('checks visible() against the search index', async function () {
    await openTab();
    await browser.executeObsidian(({ app }) => {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
      tab.__original = tab.getSettingDefinitions.bind(tab);
      const rows: any[] = tab.__original();
      const footer = rows.filter((r) => r.control?.key?.startsWith('backlinks') && r.control.key !== 'backlinksFooter');
      const rest = rows.filter((r) => !footer.includes(r));
      tab.getSettingDefinitions = () => [
        ...rest,
        { type: 'group', heading: 'Backlinks', visible: () => false, items: footer },
      ];
      tab.update();
    });
    record('search-hidden-group-backlinks', await search('backlinks'));
    await browser.executeObsidian(({ app }) => {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
      const rows: any[] = tab.__original();
      const footer = rows.filter((r) => r.control?.key?.startsWith('backlinks') && r.control.key !== 'backlinksFooter');
      const rest = rows.filter((r) => !footer.includes(r));
      tab.getSettingDefinitions = () => [
        ...rest,
        { type: 'group', heading: 'Backlinks', items: footer.map((r) => ({ ...r, control: { ...r.control, disabled: () => true } })) },
      ];
      tab.update();
    });
    await closeSettings();
    await openTab();
    await browser.executeObsidian(({ app }) => {
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      const tab = (app as any).setting.pluginTabs.find((t: any) => t.id === 'true-outliner');
      const content = tab.containerEl.closest('.vertical-tab-content') ?? tab.containerEl;
      content.scrollTop = content.scrollHeight;
    });
    await shot('disabled-group');
    record('search-disabled-group-backlinks', await search('backlinks'));
    await restoreTab();
    await closeSettings();
  });
});
