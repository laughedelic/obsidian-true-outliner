import { browser } from '@wdio/globals';
import * as fs from 'node:fs';
import * as os from 'node:os';
import * as path from 'node:path';
import * as h from '../helpers.js';

const OUT = path.join(os.tmpdir(), 'cdp-probe-') + (h.IS_MOBILE_RUN ? 'mobile' : 'desktop') + '.json';
const log: Record<string, unknown> = {};
const save = () => fs.writeFileSync(OUT, JSON.stringify(log, null, 2));
const t = async <T>(f: () => Promise<T>): Promise<[T, number]> => {
  const s = performance.now();
  const v = await f();
  return [v, +(performance.now() - s).toFixed(1)];
};

describe('cdp probe', function () {
  it('targets, handle, socket', async function () {
    const addr = (browser.capabilities as any)['goog:chromeOptions'].debuggerAddress as string;
    log.addr = addr;
    log.node = process.version;
    log.hasWS = typeof WebSocket;
    const [list, listMs] = await t(async () => (await (await fetch('http://' + addr + '/json')).json()) as any[]);
    log.targets = list.map((x) => ({ id: x.id, type: x.type, title: x.title, url: x.url.slice(0, 60) }));
    log.listMs = listMs;
    log.handle = await browser.getWindowHandle();
    log.handles = await browser.getWindowHandles();
    log.version = await (await fetch('http://' + addr + '/json/version')).json();
    save();
  });

  it('commands', async function () {
    await h.createNote('probe.md', '- a\n- b\n');
    await h.setCursor(1, 3);
    const addr = (browser.capabilities as any)['goog:chromeOptions'].debuggerAddress as string;
    const list = (await (await fetch('http://' + addr + '/json')).json()) as any[];
    const page = list.find((x) => x.type === 'page');
    const [ws, connectMs] = await t(
      () =>
        new Promise<WebSocket>((res, rej) => {
          const w = new WebSocket(page.webSocketDebuggerUrl);
          w.onopen = () => res(w);
          w.onerror = (e) => rej(new Error('ws error'));
        }),
    );
    log.connectMs = connectMs;
    let id = 0;
    const pending = new Map<number, (m: any) => void>();
    const events: any[] = [];
    ws.onmessage = (ev) => {
      const m = JSON.parse(String(ev.data));
      if (m.id && pending.has(m.id)) pending.get(m.id)!(m);
      else events.push(m);
    };
    const send = (method: string, params: object = {}) =>
      new Promise<any>((res) => {
        const n = ++id;
        pending.set(n, res);
        ws.send(JSON.stringify({ id: n, method, params }));
      });

    const [ev, evMs] = await t(() =>
      send('Runtime.evaluate', { expression: 'JSON.stringify([innerWidth, innerHeight, devicePixelRatio, navigator.maxTouchPoints, document.hasFocus()])', returnByValue: true }),
    );
    log.eval = ev.result?.result?.value;
    log.evalMs = evMs;

    const before = await h.getBuffer();
    const [k1, kMs] = await t(async () => {
      await send('Input.dispatchKeyEvent', { type: 'keyDown', key: 'x', code: 'KeyX', windowsVirtualKeyCode: 88, text: 'x' });
      return send('Input.dispatchKeyEvent', { type: 'keyUp', key: 'x', code: 'KeyX', windowsVirtualKeyCode: 88 });
    });
    await browser.pause(150);
    log.keyBefore = before;
    log.keyAfter = await h.getBuffer();
    log.keyMs = kMs;
    log.keyReply = k1;

    const [shot, shotMs] = await t(() => send('Page.captureScreenshot', { format: 'png' }));
    const png = Buffer.from(shot.result.data, 'base64');
    log.shot = { ms: shotMs, w: png.readUInt32BE(16), h: png.readUInt32BE(20), bytes: png.length };
    const [clip, clipMs] = await t(() => send('Page.captureScreenshot', { format: 'png', clip: { x: 0, y: 0, width: 200, height: 100, scale: 2 } }));
    const cp = Buffer.from(clip.result.data, 'base64');
    log.clip = { ms: clipMs, w: cp.readUInt32BE(16), h: cp.readUInt32BE(20) };

    // event subscription: console output from a WebDriver-side script
    await send('Runtime.enable');
    await browser.executeObsidian(() => console.log('probe-event'));
    await browser.pause(100);
    log.consoleEvents = events.filter((e) => e.method === 'Runtime.consoleAPICalled').length;

    // wdio still works with the socket open, and after it closes
    log.wdioWhileOpen = await browser.execute(() => 1 + 1);
    const [, closeMs] = await t(
      () => new Promise<void>((res) => { ws.onclose = () => res(); ws.close(); }),
    );
    log.closeMs = closeMs;
    log.wdioAfterClose = await browser.execute(() => 2 + 2);
    log.state = ws.readyState;
    save();
  });
});
