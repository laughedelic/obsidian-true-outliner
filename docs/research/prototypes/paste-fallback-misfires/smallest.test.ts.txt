import { it } from 'vitest';
import * as F from '../zz-f/ops';
import * as G from '../zz-g/ops';
import { parse } from '../src/parse';
import { encode } from '../src/encode';
import { walkNodes } from '../src/model';
const sh = (n: any): string => `${n.kind}[${n.lines.map((l: string) => l.trim()).join('|')}](${n.children.map(sh).join(',')})`;
const byLine = (d: any, l: string) => [...walkNodes(d)].find((n: any) => n.lines[0] === l);
it('min', () => {
  const out: string[] = [];
  const W = ['', ' ', '  ', '   ', '    ', '\t', ' \t', '\t ', '\t  ', '\t\t', '      ', '        '];
  const T = ['> q', '| a |', '2. x', 'text', '- k', '- n', '1. m'];
  const dests = [['- a\n  1. b\n', '  1. b'], ['- a\n', '- a'], ['- a\n\t- b\n', '\t- b'], ['- a\n  - b\n    - c\n', '    - c']];
  for (const [doc, anchor] of dests)
  for (const w1 of W) for (const t1 of T) for (const w2 of W) for (const t2 of T) {
    const payload = `- p\n${w1}${t1}\n${w2}${t2}\n`;
    const pp = parse(payload).children; if (pp.length !== 1) continue;
    const want = sh(pp[0]).slice(sh(pp[0]).indexOf(']'));
    const run = (O: any) => { (globalThis as any).__fb = 0; const d = parse(doc); const r = O.insertSubtrees(d, byLine(d, anchor)!.id, parse(payload).children, 'after'); const t = r.ok ? encode(r.value.doc) : 'REJ'; const pn = [...walkNodes(parse(t))].find((n: any) => /- p$/.test(n.lines[0])); return { t, fb: (globalThis as any).__fb, ok: !!pn && sh(pn).slice(sh(pn).indexOf(']')) === want }; };
    const f = run(F);
    if (!f.fb || f.ok) continue;
    const g = run(G);
    out.push(JSON.stringify({ doc, payload, fix: f.t, with270: g.t, ok270: g.ok }));
  }
  require('node:fs').writeFileSync('/tmp/claude-0/-home-user-obsidian-true-outliner/ec8969fc-18f3-54cf-81ff-87564f580056/scratchpad/min2.txt', out.length + '\n' + out.sort((a, b) => a.length - b.length).slice(0, 10).join('\n'));
});
